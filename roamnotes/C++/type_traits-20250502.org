#+title: type_traits

* Overview
- ~std::is_integral<T>::value~: Checks if the type ~T~ is an integer type.

* Implementation
#+begin_src cpp
template <typename T>
class IsIntegral {
  public:
    static const bool value{false};
};

template <>
class IsIntegral<int> {
  public:
    static const bool value{true};
};

template <>
class IsIntegral<bool> {
  public:
    static const bool value{true};
};

#+end_src

** Simplified Helper ~std::true_type~
#+begin_src cpp
#include <type_traits>

template <typename T>
struct IsIntegral : std::false_type {};

template <>
struct IsIntegral<bool> : std::true_type {};

template <>
struct IsIntegral<char> : std::true_type {};
#+end_src

template <typename T>

is_integral_v{std::is_integral<T>::value};
