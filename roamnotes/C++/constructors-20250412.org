#include <iostream>

// Simple example of copy and move contructors
class A {
public:
  A() { std::cout << "Default constructor\n"; }
  A(const A &) { std::cout << "Copy constructor\n"; }
  A(A &&) noexcept { std::cout << "Move constructor\n"; }
  A &operator=(const A &) {
    std::cout << "Copy assignment\n";
    return *this;
  }
  A &operator=(A &&) noexcept {
    std::cout << "Move assignment\n";
    return *this;
  }
  virtual ~A() { }
};

int main() {
  A a; // Default constructor
  A a2{a}; // or `A a2 = a;` Copy constructor
  A a3{std::move(a)};  // or `A a3 = std::move(a);` // Move constructor
  A a4; // Default constructor
  a4 = a2;            // Copy assignment
  a4 = A();           // Move assignment
  a4 = std::move(a2); // Move assignment
}

* Copy Constructor
- When to use copy constructor?
  - When there is a pointer member variable
  - default: shallow copy; copy pointer, not the data
- Use private default constructor to prevent default constructor
