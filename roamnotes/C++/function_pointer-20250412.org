#+title: Function Pointer

* Overview
- ~return-type (*pointer-name)([args]);~
#+begin_src cpp
#include <iostream>
int add(int a, int b) { return a + b; }

int main() {
  int (*add_ptr)(int, int) = add; // function pointer
  auto *add_ptr2 = add;           // function pointer using auto
  std::cout << "add_ptr(...): " << (unsigned long long)add_ptr << std::endl;
  std::cout << "(*add_ptr)(...): " << (unsigned long long)(*add_ptr)
            << std::endl;
  std::cout << "Result: " << add_ptr(1, 1) << std::endl;
  std::cout << "Result: " << (*add_ptr)(2, 2) << std::endl;
  std::cout << "Result: " << add_ptr2(3, 3) << std::endl;
}
#+end_src

#+RESULTS:
| add_ptr(...):    | 94181223887209 |
| (*add_ptr)(...): | 94181223887209 |
| Result:          |              2 |
| Result:          |              4 |
| Result:          |              6 |
- Direct syntax: ~add_ptr(2, 2)~~
  - The direct syntax is more commonly used because it's cleaner.
- Dereference syntax: ~(*add_ptr)(2, 2)~
  - The dereference syntax is more explicit about what's happening.

* Alias to Function Pointer
#+begin_src cpp
typedef int (*FuncPtr)(int, int);
using FuncPtr2 = int (*)(int, int);
int add(int a, int b) { return a + b; }
FuncPtr2 add_ptr = add; // function pointer
#+end_src
