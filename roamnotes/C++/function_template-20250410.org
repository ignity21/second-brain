#+title: Function Template

* Overview
- Declaration
#+begin_src cpp
// generally preferred
template <typename T, typename U> void func(T t, U u) {};
// for template template parameters
template <class T, class U> void func2(T t, U u) {};
#+end_src
- Instantiation
#+begin_src cpp
// Explicit instantiation
template void func<float, double>(float t, double u); // Explicit instantiation

// Implicit instantiation: void func<int, double>(int, double)
func<int, double>(1, 2.0);

#+end_src

* Specialization
Overriding the template function.
#+begin_src cpp
template<typename T = int> // default template argument
T max(T a, T b) {
  return (a > b) ? a : b;
};

// explicit instantiation
template long max<long>(long a, long b);

// explicit specialization, should provide an implementation
template<> int max<int>(int a, int b) {...};
// <> indicates you are providing a specialization
#+end_src

* Variadic Arguments
#+begin_src cpp
template<typename... Args>
void func(Args... args) {
  std::cout << sizeof...(args) << std::endl;
}
#+end_src
