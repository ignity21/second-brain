* ~throw~

  throw "char prointer exception";

#+end_src
- *Recommended* to throw ~std::exception~ or its derived classes

* Custom Exceptions
#+begin_src cpp
class my_exception : public std::exception {
  unsigned int code;
  string reason;

public:
  my_exception(unsigned int c, const string &r) : code(c), reason(r) {}
  virtual const char *what() const noexcept override { return reason.c_str(); }
};

int main() {
  try {
    throw my_exception{1, "Error occurred"};
  } catch (const my_exception &e) { // or its base classes std::exception
    cout << e.code << ": " << e.reason << endl;
  }
  return 0;
}
#+end_src
