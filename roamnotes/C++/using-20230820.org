* Using Type Aliases
#+begin_src cpp
  typedef std::unique_ptr<std::unordered_map<std::string, int>> UPtrMapSI;
  using UPtrMapSI = std::unique_ptr<std::unordered_map<std::string, int>>;

  // function pointers
  typedef void (*FP)(int, const std::string&);
  using FP = void (*)(int, const std::string&);
#+end_src

* Templatized Alias Declaration
  #+begin_src cpp
    template<typename T>
    using MyAllocList = std::list<T, MyAlloc<T>>;

    // client code
    MyAllocList<Widget> lw;
  #+end_src
- Alternative: ~typedef~ version
#+begin_src cpp
  template<typename T>
  struct MyAllocList {
    typedef std::list<T, MyAlloc<T>> type;
  };

  // client code
  MyAllocList<Widget>::type lw;
#+end_src
- Alias templates avoid the "::type" suffix

* Type Traits Transformations in C11 and C14
  #+begin_src cpp
    std::remove_const<T>::type // C++11: const T -> T, historical reason
    std::remove_const_t<T> // C++14, preferred one.

    // C++14 implementation
    template <class T>
    using remove_const_t = typename remove_const<T>::type;
  #+end_src
- same for ~remove_reference_t~, ~add_lvalue_reference_t~
