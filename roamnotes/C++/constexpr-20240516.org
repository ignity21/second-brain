#+title: ~constexpr~
#+filetags: C++

* ~constexpr~ on Variable
#+begin_src cpp
const int a{42};
constexpr int b{a+3};
#+end_src

* ~constexpr~ Function
#+begin_src cpp
constexpr int multiply(int a, int b) { return a * b; }

int main() {
  constexpr int c = multiply(3, 4); // constexpr int c = 12;
  int d = multiply(c, 11);  // int d = 132;

  // if the arguments are not const, the function is evaluated at run-time
  int f = multiply(d, 5);
  return 0;
}
#+end_src
- *Evaluate at run-time* if the arguments are not constexpr
  - Can be verified by using the *debugger*
- ~constexpr~ function can't call a non-constexpr function
- ~constexpr~ function can not throw an exception
- ~constexpr~ function can be recursive, depth can be limited by option ~-fconstexpr-depth=...~~


#+begin_src cpp
constexpr int len_foo() { return 4; }

// recursive constexpr function
constexpr int fibonacci(const int n) {
  return n <= 1 ? n : fibonacci(n - 1) + fibonacci(n - 2);
}

// C++14 with loops and if statements
constexpr int fibonacci14(const int n) {
  if (n == 0) return 0;
  if (n == 1) return 1;
  return fibonacci14(n - 1) + fibonacci14(n - 2);
}
#+end_src

* ~constexpr~ on Class
- User-defined literal types
- Defined by using ~constexpr~ constructor
#+begin_src cpp
class Complex {
public:
  constexpr Complex(double r, double i) : re(r), im(i) {}
  constexpr double real() const { return re; }
  constexpr double imag() const { return im; }

private:
  double re;
  double im;
};

int main() {
  constexpr Complex c1(3.0, 4.0);
  constexpr Complex c2(1.0, 2.0);
  return 0;
}
#+end_src

* ~if constexpr~ (C++17)
#+begin_src cpp
template<typename T>
auto increment(const T& t) {
    if constexpr (std::is_integral_v<T>) {
        return t + 1;
    } else {
        return t + 0.01;
    }
}

// It will split into two functions at compile time
int increment_c(const int& t) { return t + 1; }
double increment_c(const double& t) { return t + 0.01; }
#+end_src
