#+title: ~if constexpr~

* Use Case
- Compile-time branching
  - Avoids unnecessary code generation

* Example
#+begin_src cpp
template<typename T>
auto increment(const T& t) {
    if constexpr (std::is_integral_v<T>) {
        return t + 1;
    } else {
        return t + 0.01;
    }
}
#+end_src
