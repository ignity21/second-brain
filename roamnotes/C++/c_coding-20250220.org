** Loop
*** For Loop
#+begin_src cpp
// Two indices for loop
bool isPalindrome(string str) {
  for (int i = 0, j = str.size() - 1; i < j; i++, j--) {
    if (str[i] != str[j]) {
      return false;
    }
  }
  return true;
}
#+end_src
