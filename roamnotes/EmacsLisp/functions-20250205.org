** Buffer-related
#+begin_src elisp
(defun simplified-beginning-of-buffer ()
  "Move point to the beginning of the buffer;
leave mark at previous position."
  (interactive)
  (push-mark) ; leave a mark at the original position in the buffer; C-h v mark-ring
  (goto-char (point-min)) ; move to the beginning of the buffer
  )

(defun mark-whole-buffer ()
  "Put point at beginning and mark at end of buffer.
Just toy, it is usually a mistake for a Lisp function to use any subroutine
that uses or sets the mark."
  (interactive)
  (push-mark (point)) ; same as (push-pack)
  (push-mark (point-max) nil t) ; set a mark at the end of the buffer
  (goto-char (point-min)) ; move to the beginning of the buffer
  )

(defun copy-to-buffer (buffer start end)
  "Replace the text of the current buffer with the text of the region."
  (interactive "BCopy to buffer: \nr")
  (let ((oldbuf (current-buffer))) ; bind-oldbuf-to-value-of-current-buffer
    (with-current-buffer (get-buffer-create buffer) ; with-the-buffer-you-are-copying-to
      (barf-if-buffer-read-only) ; but-do-not-erase-or-copy-to-a-read-only-buffer
      (erase-buffer)
      (save-excursion
        (insert-buffer-substring oldbuf start end) ; insert-substring-from-oldbuf-into-buffer
        ))))
#+end_src

** Text-related
#+begin_src elisp
(defun zap-to-char (arg char)
  "Kill up to and including ARG'th occurrence of CHAR.
Case is ignored if `case-fold-search' is non-nil in the current buffer.
Goes backward if ARG is negative; error if CHAR not found."
  (interactive "p\ncZap to char: ")
  ;; '\n' is the separator
  ;; 'p' means that the first argument will be passed the value of a processed prefix.
  (if (char-table-p translation-table-for-input)
      (setq char (or (aref translation-table-for-input char) char)))
  (kill-region (point) (progn
                         (search-forward (char-to-string char)
                                         nil nil arg)
                         (point))))
#+end_src
