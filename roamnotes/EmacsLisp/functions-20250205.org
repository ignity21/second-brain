* Examples
#+begin_src elisp
(defun simplified-beginning-of-buffer ()
  "Move point to the beginning of the buffer;
leave mark at previous position."
  (interactive)
  (push-mark) ; leave a mark at the original position in the buffer; C-h v mark-ring
  (goto-char (point-min)) ; move to the beginning of the buffer
  )

(defun mark-whole-buffer ()
  "Put point at beginning and mark at end of buffer.
Just toy, it is usually a mistake for a Lisp function to use any subroutine
that uses or sets the mark."
  (interactive)
  (push-mark (point)) ; same as (push-pack)
  (push-mark (point-max) nil t) ; set a mark at the end of the buffer
  (goto-char (point-min)) ; move to the beginning of the buffer
  )
#+end_src

* ~interactive~ Special Forms
- Call interactive function for its *side effects*
  - return value is not displayed in the echo area (should use ~(message)~)
- ~(interactive "p")~
  - The "p" tells Emacs to pass the prefix (number) to the function
  - Use =C-u prefix M-x= to pass arguments
#+begin_src elisp
(defun multiply-by-seven (number)       ; Interactive version.
  "Multiply NUMBER by seven."
  (interactive "p")
  (message "The result is %d" (* 7 number)))

(defun name-of-function (num char)
  "documentation..."
  (interactive "p\ncZap to char: ") ; p fornumber, c for char
  (message "num: %d, char: %c" num char))
#+end_src
