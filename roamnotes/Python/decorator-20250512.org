#+title: Decorator

* 用途
- 运行时校验，如 ~login_required~
- 注入额外参数，如 ~@unittest.mock.patch~
- 缓存，如 ~@functools.lru_cache~
- 注册函数，如 ~@app.route~
- 替换为复杂对象，如 ~@property~, ~@statcimethod~

* 函数装饰器
** 无参数
#+begin_src python :results output
import time
from functools import wraps

def timer(func):
    @wraps(func)
    def decorated(*args, **kwargs):
        st = time.perf_counter()
        ret = func(*args, **kwargs)
        print(f"time cost: {time.perf_counter() - st}s")
        return ret
    return decorated

@timer
def sleep_a_while(sec):
    time.sleep(sec)

import time
from functools import wraps

def timer(print_args=False):
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            start_time = time.time()
            result = func(*args, **kwargs)
            end_time = time.time()

            if print_args:
                print(f'"{func.__name__}", args: {args}, kwargs: {kwargs}')
            print(f"time cost: {elapsed_time}s")
            return result
        return wrapper
    return decorator

import time
from functools import wraps

def timer(func=None, *, print_args=False):
    def decorator(_func):
        @wraps(_func)
        def wrapper(*args, **kwargs):
            start_time = time.time()
            result = _func(*args, **kwargs)
            end_time = time.time()
            elapsed_time = end_time - start_time
            if print_args:
                print(f'"{func.__name__}", args: {args}, kwargs: {kwargs}')
            print(f"time cost: {elapsed_time}s")
            return result
        return wrapper
    if func is None:
        return decorator
    else:
        return decorator(func)

@timer
def sleep_a_while(sec):
    time.sleep(sec)

import time
from functools import update_wrapper, partial

class Timer:
    def __init__(self, func, *, print_args=False):
        update_wrapper(self, func)
        self._func = func
        self._print_args = print_args
        self._time_taken = 0

    def __call__(self, *args, **kwargs):
        start_time = time.time()
        result = self._func(*args, **kwargs)
        end_time = time.time()
        self._time_taken = end_time - start_time
        if self._print_args:
            print(f'"{self._func.__name__}", args: {args}, kwargs: {kwargs}')
        print(f"time cost: {self._time_taken}s")
        return result

    def get_time_taken(self):
        return self._time_taken

def timer(**kwargs):
    return partial(Timer, **kwargs)

@timer(print_args=True)
def sleep_a_while(sec):
    time.sleep(sec)

sleep_a_while(1)
print(f"get_time_taken: {sleep_a_while.get_time_taken()}")

_validators = {}
def register(cls):
    global _validators
    _validators[cls.name] = cls
    return cls

def get_validator(name):
    return _validators.get(name)

@register
class StringValidator:
    name = "string"

@register
class IntegerValidator:
    name = "int"

print(get_validator("string"))

* 用 [[id:ffb4d5cb-79fe-4565-a40a-a8ef50de415a][wrapt]] 装饰类成员函数
- 自己实现需要区分是普通函数还是类成员函数
#+begin_src python :results output
import time
import wrapt

def timer(print_args=False):
    def decorator(func):
        @wrapt.decorator
        def wrapper(wrapped, instance, args, kwargs):
            # wrapped: the original function or method
            # instance:
            # - `instance` for `class instance`
            # - `class` for classmethod
            # - None for functions(including staticmethod)
            start_time = time.time()
            result = wrapped(*args, **kwargs)
            end_time = time.time()
            elapsed_time = end_time - start_time
            if print_args:
                print(f'"{wrapped.__name__}", args: {args}, kwargs: {kwargs}')
            print(f"time cost: {elapsed_time}s")
            return result
        return wrapper(func)
    return decorator

class Sleeper:
    def __init__(self, sec):
        self._sec = sec

    @timer(print_args=True)
    def sleep_a_while(self):
        time.sleep(self._sec)

sleeper = Sleeper(3)
sleeper.sleep_a_while()
