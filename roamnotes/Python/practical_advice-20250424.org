* 容器相关
** ~namedtuple~
- 多用于函数的返回值
#+begin_src python

#+end_src
- 带类型注解 ~NamedTuple~
#+begin_src python

class Rectangle(NamedTuple):
    width: int
    height: int
#+end_src

** 拷贝可变类型
- ~copy.copy()~ 浅拷贝
#+begin_src python
from copy import copy
org = [1, 2, 3]

# shallow copied
c1 = copy(org)
c2 = org.copy()
c3 = [x for x in org]
#+end_src
- ~copy.deepcopy()~ 深拷贝：拷贝嵌套对象
#+begin_src python
from copy import copy, deepcopy
org = [1, 2, [3, 4]]

c1 = copy(org) # shallow copy
assert id(c1[-1]) == id(org[-1])
deepcopied = deepcopy(org)
assert id(deepcopied[-1]) != id(org[-1])
#+end_src
** 解包容器对象
#+begin_src python :results output
l = [1, 2, *range(3, 10)]

d1 = {"a": 1}

d_merged = {**d1, **d2}

** 使用 ~bisect~ 将数值转换为 Categorical
#+begin_src python
from bisect import bisect
breakpoints = [6, 7, 8, 8.5]
grades = ["D", "C", "B", "A", "S"]

def get_grade(score):
    index = bisect(breakpoints, score)
    return grades[index]

return get_grade(8.4)
#+end_src

#+RESULTS:
: A
