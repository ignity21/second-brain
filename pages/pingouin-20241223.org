#+title: pingouin

* t-test Example
** Paired Sample t-test
#+begin_src python
import pingouin

pingouin.ttest(x=sample_data['diff'],
               y=0, # null hypothesis
               alternative='less', # left-tailed test
               )
# alternative: 'less', 'greater', 'two-sided'
#+end_src
or
#+begin_src python
pingouin.ttest(x=sample_data['2021'],
               y=sample_data['2024'],
               paired=True,
               alternative='less')
#+end_src
