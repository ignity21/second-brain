* Simple Lock-free Examples
** Atomic Queue
#+begin_src c++
int q[N];
std::atomic<size_t> front{0};
void push(int x) {
    size_t pos = front.fetch_add(1);
    q[pos] = x; // exclusive slot
    // ignoring the synchronization with the reads in the example
}
#+end_src

** Atomic List
#+begin_src c++
struct node {int value; node *next; };
std::atomic<node*> head;
void push_front(int x) {
    node *new_node = new node;
    new_node->value = x;
    node *old_head = head;
    do { new_node->next = old_head; }
    while (!head.compare_exchange_strong(old_head, new_node));
    // if head == old_head, head = new_node
    // else old_head = head, then new_node->next = old_head
}
#+end_src
