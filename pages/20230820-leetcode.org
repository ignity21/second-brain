#+begin_src cpp
  void swap(int& a, int& b) {
    a ^= b; // a: bit位为1的位置表示a和b在该位上不同，bit位为0则表示相同位
    b ^= a;
    // b: ab相同位为0 b与相同位0异或原bit位值不变 1^0=1 0^0=0
    // b: ab不同位为1 b与不同位1异或原bit位反转 1^1=0 0^1=1
    a ^= b; // 与第二步同理
  }
#+end_src
