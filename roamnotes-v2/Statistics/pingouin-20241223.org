#+title: pingouin

* t-test Example
** Paired Sample t-test
#+begin_src python
import pingouin

pingouin.ttest(x=sample_data['diff'],
               y=0, # null hypothesis
               alternative='less', # left-tailed test
               )
# alternative: 'less', 'greater', 'two-sided'
#+end_src
or
#+begin_src python
pingouin.ttest(x=sample_data['2021'],
               y=sample_data['2024'],
               paired=True,
               alternative='less')
#+end_src

* ANOVA Example
** One-way
#+begin_src python
pingouin.anova(data=stack_overflow,
               dv='converted_comp',
               between='job_sat', # categorical variable
               )
# p-value stored in 'p-unc' column

# to find out which group is different from the others
pingouin.pairwise_tests(data=stack_overflow,
                        dv='converted_comp',
                        between='job_sat',
                        padjust='none', # no correction
                        )
#+end_src
