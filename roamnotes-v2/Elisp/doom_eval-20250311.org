#+title: Code Evaluation in Doom Emacs

* Register a REPL for a major-mode
- ~+eval-repls~: list of available REPLs
- ~(set-repl-handler! MAJOR-MODES FUNCTION)~
- Example:
#+begin_src elisp
(defun +lua/open-repl ()
  (interactive)
  (lua-start-process "lua" "lua")
  (pop-to-buffer lua-process-buffer))

(set-repl-handler! 'lua-mode #'+lua/open-repl)
#+end_src

* Code Evaluation Handler
- Run regions or entire buffers with *Quickrun*
- ~quickrun--language-alist~
- Example:
#+begin_src elisp
(set-eval-handler! 'groovy-mode "groovy")
(set-eval-handler! 'crystal-mode
  '((:command     . "crystal")
    (:exec        . "%c %s")
    (:description . "Run Crystal script")))
(set-eval-handler! 'emacs-lisp-mode #'+emacs-lisp-eval)
#+end_src
