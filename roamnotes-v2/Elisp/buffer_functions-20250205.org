** ~(insert-buffer-substring)~
- ~(insert-buffer-substring BUFFER &optional START END)~
Copy a region of text from a buffer that is passed to the function as an argument and insert the region into the current buffer.

* Examples
#+begin_src elisp
(defun simplified-beginning-of-buffer ()
  "Move point to the beginning of the buffer and leave mark at the old position."
  (interactive)
  (push-mark)
  (goto-char (point-min)))

(defun copy-to-buffer (buffer start end)
  "Replace BUFFER's text with the region from START to END."
  (interactive "BCopy to buffer: \nr")
  (let ((source-buffer (current-buffer)))
    (with-current-buffer (get-buffer-create buffer)
      (barf-if-buffer-read-only)
      (erase-buffer)
      (insert-buffer-substring source-buffer start end))))
#+end_src
